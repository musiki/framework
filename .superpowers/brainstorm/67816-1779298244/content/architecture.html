Tabla de snapshots (Postgres)

CREATE TABLE note_snapshots (
  id SERIAL PRIMARY KEY,
  course_id TEXT NOT NULL,
  file_path TEXT NOT NULL, -- relativo a src/content/
  content TEXT NOT NULL, -- el .md completo
  saved_by TEXT NOT NULL, -- email del teacher
  saved_at TIMESTAMPTZ DEFAULT now(),
  op TEXT -- 'save' | 'create' | 'delete' | 'move'
);
